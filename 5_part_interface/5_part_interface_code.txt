package notifire

type Notifire interface{ Notify(message string) }

func SendNotification(n Notifire, message string) { n.Notify(message) }

=== cmd/main.go ===
package main

import (
	"go_restart/5_part_interface/internal/console"
	"go_restart/5_part_interface/internal/email"
	"go_restart/5_part_interface/internal/notifire"
	"go_restart/5_part_interface/internal/sms"
	"go_restart/5_part_interface/internal/telegram"
)

func main() {
	notifires := []notifire.Notifire{
		console.ConsoleNotify{},
		email.EmailNotificator{},
		sms.SmsNotificator{},
		telegram.TelegramNotificator{},
	}

	for _, v := range notifires {
		notifire.SendNotification(v, "ваш заказ готов")
	}
}
